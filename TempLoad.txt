def calculate_category_averages(self):
    """Calculate average scores for each category"""
    category_averages = {}
    
    for category in self.categories:
        # Get all questions in this category
        category_questions = self.questions_df[self.questions_df['Category'] == category]
        total_score = 0
        answered_questions = 0
        
        for q_id in category_questions['QuestionID']:
            if q_id in self.lm_responses and self.lm_responses[q_id] is not None:
                total_score += self.lm_responses[q_id]
                answered_questions += 1
        
        # Calculate average if there are answered questions
        if answered_questions > 0:
            average = total_score / answered_questions
            category_averages[category] = round(average, 2)
        else:
            category_averages[category] = None
    
    return category_averages

def on_lm_radio_toggled(self):
    """Handle LM feedback radio button selections and update tab averages"""
    sender = self.sender()
    if sender.isChecked():
        _, q_id, value = sender.objectName().split('_')
        self.lm_responses[q_id] = int(value)
        self.update_tab_titles_with_averages()

def update_tab_titles_with_averages(self):
    """Update tab titles to show category averages"""
    averages = self.calculate_category_averages()
    
    for i in range(self.lm_tabs.count()):
        category = self.categories[i]
        avg = averages[category]
        
        if avg is not None:
            self.lm_tabs.setTabText(i, f"{category} (Avg: {avg})")
        else:
            self.lm_tabs.setTabText(i, f"{category} (No answers)")

def create_lm_feedback_page(self):
    """Create the tabbed LM feedback interface"""
    lm_page = QWidget()
    lm_layout = QVBoxLayout(lm_page)
    
    # Create tabs for LM feedback
    self.lm_tabs = QTabWidget()
    self.lm_responses = {}
    
    # Get unique categories from the Excel file
    self.categories = self.questions_df['Category'].unique().tolist()
    
    for category in self.categories:
        scroll = QScrollArea()
        scroll.setWidgetResizable(True)
        
        container = QWidget()
        layout = QVBoxLayout(container)
        
        # Filter questions for this category
        category_questions = self.questions_df[self.questions_df['Category'] == category]
        
        for _, row in category_questions.iterrows():
            q_id = row['QuestionID']
            question_text = row['Question']
            
            group_box = QGroupBox(question_text)
            group_layout = QVBoxLayout()
            
            # Create radio buttons for options
            option_group = QButtonGroup(self)
            self.lm_responses[q_id] = None
            
            options = [row['Option1'], row['Option2'], row['Option3'], row['Option4']]
            option_values = [1, 2, 3, 4]  # Numeric values for options
            
            for i, (option, value) in enumerate(zip(options, option_values)):
                radio = QRadioButton(option)
                radio.setObjectName(f"lm_{q_id}_{value}")
                radio.toggled.connect(self.on_lm_radio_toggled)
                option_group.addButton(radio)
                group_layout.addWidget(radio)
            
            group_box.setLayout(group_layout)
            layout.addWidget(group_box)
        
        scroll.setWidget(container)
        self.lm_tabs.addTab(scroll, f"{category} (No answers)")
    
    lm_layout.addWidget(self.lm_tabs)
    self.stacked_widget.addWidget(lm_page)
