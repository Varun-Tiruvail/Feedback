def update_analysis(self):
    """Update the analysis chart with current data"""
    self.fig.clear()
    
    # Calculate your averages
    your_averages = self.calculate_category_averages()
    
    # Get team averages (placeholder - replace with actual data from your DB)
    team_averages = {
        cat: 2.5 for cat in self.categories  # Example data - replace with real team averages
    }
    
    # Prepare data for plotting
    categories = list(your_averages.keys())
    your_scores = [your_averages[cat] or 0 for cat in categories]
    team_scores = [team_averages.get(cat, 0) for cat in categories]
    
    # Create plot
    ax = self.fig.add_subplot(111)
    width = 0.35
    x = range(len(categories))
    
    # Plot bars
    rects1 = ax.bar([i - width/2 for i in x], your_scores, width, 
                   label='Your Scores', color='#1f77b4')
    rects2 = ax.bar([i + width/2 for i in x], team_scores, width, 
                   label='Team Averages', color='#ff7f0e')
    
    # Add labels and formatting
    ax.set_ylabel('Average Score')
    ax.set_title('Your Scores vs Team Averages')
    ax.set_xticks(x)
    ax.set_xticklabels(categories, rotation=45, ha='right')
    ax.legend()
    
    # Add value labels on top of bars
    for rect in rects1 + rects2:
        height = rect.get_height()
        if height > 0:
            ax.text(rect.get_x() + rect.get_width()/2., height,
                   f'{height:.1f}', ha='center', va='bottom')
    
    # Adjust layout and draw
    self.fig.tight_layout()
    self.canvas.draw()
